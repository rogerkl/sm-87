# The sampler's pages: EDIT, ENV, LFO; SELECT the slot, K1..K4 the values; SAVE keeps them
wait 600
tap ENV
wait 30
expect page 5
# FILTER of slot 1, from 99 down 9
turn K1 -9
wait 30

expect unsaved 1
shot out.png

turn SELECT 1

expect slot 2

wait 30
